1064 bool hasConstructor = std::is_default_constructible<T>::value && !std::has_trivial_default_constructor<T>::value;
1065 bool hasDestructor = std::is_destructible<T>::value && !std::is_trivially_destructible<T>::value;

1067 bool hasCopyConstructor = std::is_copy_constructible<T>::value && !std::has_trivial_copy_constructor<T>::value;
1068#else
1069 // All other compilers and versions are assumed to use non C++11 compliant code until proven otherwise
1070 // Not fully C++11 compliant. The has_trivial checks were used while the standard was still
1071 // being elaborated, but were then removed in favor of the above is_trivially checks
1072 // http://stackoverflow.com/questions/12702103/writing-code-that-works-when-has-trivial-destructor-is-defined-instead-of-is
